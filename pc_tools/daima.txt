module ADC_CAPTURE_DATA(
	
		input ad_bck,
		input ad_data,
		input rst_n,
		input ad_lrck,
		
		output reg [23:0] rdldat1,
		output reg [23:0] rdldat2,
		
		output reg ldat_rdy1,
		output reg ldat_rdy2,
		
		
		output wire trig_sig_ok
		

    );

	reg [31:0] bck_lcnt1 = 32'd0;
	reg [31:0] bck_lcnt2 = 32'd0;
	
	
	always @(posedge ad_bck or negedge rst_n )
	if(!rst_n) begin	
		bck_lcnt1 <=0;
		ldat_rdy1 <= 0;
		rdldat1 <= 24'h112233;	

		bck_lcnt2 <=0;
		ldat_rdy2 <= 0;
		rdldat2 <= 24'h112233;	
				
	end
	else begin			
		if(ad_lrck)
			begin
				
				bck_lcnt2 <= 0;		
				ldat_rdy2 <= 0;
				
				if(bck_lcnt1<24)
					begin
						ldat_rdy1 <= 0;
						bck_lcnt1 <= bck_lcnt1 + 1;
						rdldat1 <= {rdldat1[22:0],ad_data};//
					end
				else 
					begin
						bck_lcnt1 <= 0;
						ldat_rdy1 <= 1;
					end
			end
		else if(~ad_lrck)
			begin
				bck_lcnt1 <= 0;		
				ldat_rdy1 <= 0;
				
				begin
				
					if(bck_lcnt2<24)
						begin
							ldat_rdy2 <= 0;
							bck_lcnt2 <= bck_lcnt2 + 1;
							rdldat2 <= {rdldat2[22:0],ad_data};//
						end
					else 
						begin
							bck_lcnt2 <= 0;
							ldat_rdy2 <= 1;
						end
				end
				
			end
		
	end


	
	
	
	
	
	

(* noprune *) reg yuzhi = 1'b0;
	
	
	// 定义16位有符号寄存器存储ADC数据
reg signed [23:0] adc_data_signed;
reg signed [23:0] adc_data_signed_r1;

// 同步打拍：将输入数据寄存，优化FPGA时序（标准设计方法）
always @(posedge ad_bck or negedge rst_n) begin
    if(!rst_n) begin
        adc_data_signed <= 24'sd0;     // 复位：有符号数0
		  adc_data_signed_r1 <= 24'sd0; 
    end
    else if(ldat_rdy1)
		begin
        adc_data_signed <= rdldat1; // 采样ADC数据
		  adc_data_signed_r1 <= adc_data_signed ;
		end
end

// 阈值判断逻辑：>100 或 < -100 时标志拉高
always @(posedge ad_bck or negedge rst_n) begin
    if(!rst_n) begin
        yuzhi <= 1'b0;        // 复位：标志位拉低
    end
    else begin
        // 16位有符号数直接比较补码大小
        if((adc_data_signed > 24'sd30000 || adc_data_signed < -24'sd30000) && (adc_data_signed_r1 > 24'sd30000 || adc_data_signed_r1 < -24'sd30000))
				begin
					yuzhi <= 1'b1;    // 超限，标志置1
				end
        else begin
            yuzhi <= 1'b0;    // 正常，标志置0
        end
    end
end
	
	
	assign trig_sig_ok = yuzhi;

reg trig_sig;

always @(posedge ad_bck or negedge rst_n) begin
    if (!rst_n) begin
        trig_sig <= 1'b0;  // 异步复位
    end else begin
        // 条件：正数 > 9000  或者  负数 < -9000
        if ( ((~adc_right[15]) && (adc_right[14:0] > 15'd9000)) ||    // 正数超阈值
              (adc_right[15]  && (adc_right[14:0] < 15'd27536)) )    // 负数 < -9000
        begin 
            trig_sig <= 1'b1;
        end 
        else begin 
            trig_sig <= 1'b0;
        end
    end
end

									ram_wr_addr <= ram_wr_addr;
								end

									state <= S1;
									yuzhi_point_addr <= ram_wr_addr ; //如果检测到阈值之后，将当前的地址记住
									max_value_group2 <= 16'd0;
								end

								end 
						end 
						
					S1: //在这个状态中继续写入数据 2048 -200 = 1848

							if(adc_valid) //当数据到达的时候，一个个的写入到RAM缓存中

									ram_wr_addr <= ram_wr_addr;
								end 
							
							if((yuzhi_point_wr_cnt == 1947) && adc_valid)//2048-1-200
								begin 
									state <= S2;
									yuzhi_point_wr_cnt <= 16'd0;
								end 
							else if(adc_valid)
								begin 
									yuzhi_point_wr_cnt <= yuzhi_point_wr_cnt + 1'b1 ;
								end 
						end 
						
					S2://将使能信号关闭，关闭当前写操作
						begin 
							ram_wr_en <= 1'b0;
							state <= S3;
							
							max_value_group2 <= max_value_group2;
							
							yuzhi_point_wr_cnt <= 16'd0;
							
							//adc_ram_rdaddress <= yuzhi_point_addr - 200; //读取RAM的地址时候，将地址倒退200个单位
						end 
						
					S3: //将当前RAM的数据写入到另一个RAM中，另一个RAM的深度是2048
						begin ///////产生一个标志信号，拉高300个时钟周期
						//	adc_ram_rden <= 1'b1;
							//adc_ram_rdaddress <= adc_ram_rdaddress + 1'b1;
							
							//加入查找最大数值的逻辑
							
							
							if(delay_cnt == 16'd10)
								begin 
									delay_cnt <= 16'd0;
									state <= S4;
									ok_read_flag <= 1'b0;
								end 
							else 
								begin 
									delay_cnt <= delay_cnt + 1'b1 ;
								end 
							
							
							ok_read_flag <= 1'b1;
							
							
//							if(max_value < adc_left)

						//	adc_ram_rdaddress <= 12'd0;
							
							state <= IDLE;
					
					
						end 
					
				endcase 
			end 
	end
