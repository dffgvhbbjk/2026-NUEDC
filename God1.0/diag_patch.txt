void SerialWorker::onReadyRead()
{
    QByteArray data = m_serial->readAll();
    m_rxTotal += data.size();
    emit bytesReceived(m_rxTotal);

    static int s_rxCount = 0;
    s_rxCount++;
    if (s_rxCount <= 10 || s_rxCount % 100 == 0) {
        qDebug() << "[DIAG-RX] #" << s_rxCount
                 << "read" << data.size() << "bytes, total=" << m_rxTotal
                 << "thread=" << QThread::currentThreadId();
    }

    m_parser.feed(data);
}
